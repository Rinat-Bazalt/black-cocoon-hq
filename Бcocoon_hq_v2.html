<script>

  function switchTab(name) {

    document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));

    document.querySelectorAll('.panel').forEach(p => p.classList.remove('active'));

    document.getElementById(name).classList.add('active');

    document.querySelector(`.tab[onclick*="${name}"]`).classList.add('active');

    if (name === '3d') init3D();

  }


  function printPanel(name) {

    const panel = document.getElementById(name);

    const clone = panel.cloneNode(true);

    const win = window.open('', '_blank');

    win.document.write('<html><head><title>BLACK COCOON - Чертеж</title></head><body style="background:white; display:flex; justify-content:center;">');

    win.document.write(clone.innerHTML);

    win.document.write('</body></html>');

    win.document.close();

    win.print();

  }


  function init3D() {

    if (window._3dready) return;

    window._3dready = true;

    const panel = document.getElementById('3d');

    panel.innerHTML = '<button class="btn-print" onclick="printPanel(\'3d\')">Печать 3D</button>';

    const script = document.createElement('script');

    script.src = 'https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js';

    script.onload = function() {

      const orbitScript = document.createElement('script');

      orbitScript.src = 'https://cdn.jsdelivr.net/npm/three@0.128.0/examples/js/controls/OrbitControls.js';

      orbitScript.onload = function() {

        const W = 14.75, L = 10.0, H = 9.44;

        const scene = new THREE.Scene(); scene.background = new THREE.Color(0x111122);

        const camera = new THREE.PerspectiveCamera(45, panel.clientWidth / panel.clientHeight, 0.1, 1000);

        camera.position.set(15, 10, 20); camera.lookAt(0, 0, 4);

        const renderer = new THREE.WebGLRenderer({ antialias: true });

        renderer.setSize(panel.clientWidth, panel.clientHeight);

        panel.appendChild(renderer.domElement);

        const controls = new THREE.OrbitControls(camera, renderer.domElement);

        controls.target.set(0, 0, 4); controls.update();

        scene.add(new THREE.AmbientLight(0x404066));

        const dirLight = new THREE.DirectionalLight(0xffffff, 0.8); dirLight.position.set(10, 20, 15); scene.add(dirLight);

        function addLine(p1, p2, color) {

          const g = new THREE.BufferGeometry();

          g.setAttribute('position', new THREE.BufferAttribute(new Float32Array([p1.x, p1.y, p1.z, p2.x, p2.y, p2.z]), 3));

          scene.add(new THREE.Line(g, new THREE.LineBasicMaterial({ color })));

        }

        const v0 = new THREE.Vector3(-W/2, -L/2, 0), v1 = new THREE.Vector3(W/2, -L/2, 0), v2 = new THREE.Vector3(W/2, L/2, 0), v3 = new THREE.Vector3(-W/2, L/2, 0);

        const v4 = new THREE.Vector3(0, -L/2, H), v5 = new THREE.Vector3(0, L/2, H);

        addLine(v0, v1, 0xffffff); addLine(v1, v2, 0xffffff); addLine(v2, v3, 0xffffff); addLine(v3, v0, 0xffffff);

        addLine(v0, v4, 0xff4444); addLine(v1, v4, 0xff4444); addLine(v2, v5, 0xff4444); addLine(v3, v5, 0xff4444);

        addLine(v4, v5, 0x4444ff);

        scene.add(new THREE.GridHelper(15, 20, 0x333366, 0x333366));

        function animate() { requestAnimationFrame(animate); controls.update(); renderer.render(scene, camera); }

        animate();

        window.addEventListener('resize', () => { camera.aspect = panel.clientWidth / panel.clientHeight; camera.updateProjectionMatrix(); renderer.setSize(panel.clientWidth, panel.clientHeight); });

      };

      document.head.appendChild(orbitScript);

    };

    document.head.appendChild(script);

  }


  window.onload = function() { switchTab('3d'); };

</script>
